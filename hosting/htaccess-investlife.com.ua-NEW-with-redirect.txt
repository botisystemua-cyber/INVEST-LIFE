# Решта файлу — без змін, як було.
#
# Винятки (НЕ редіректяться):
#   /wp-admin      — щоб лишався доступ до адмінки
#   /wp-login.php  — форма входу
#   /wp-content    — картинки, стилі, завантаження
#   /wp-includes   — службові файли WordPress
#   /.well-known   — КРИТИЧНО: через цей шлях Let's Encrypt поновлює
#                    сертифікат. Без винятку SSL перестане поновлюватись.
#
# R=302 — тимчасовий редірект (легко відкотити).
#         Коли переконаєтесь, що все працює як треба — можна змінити на R=301.
# QSA   — зберігає параметри з рекламного посилання (utm_source тощо)
# ref=comua — мітка, за якою в аналітиці investlife.es видно трафік
#             з українського домену

# --- Редірект на іспанський сайт ---
RewriteEngine On

RewriteCond %{REQUEST_URI} !^/wp-admin
RewriteCond %{REQUEST_URI} !^/wp-login\.php
RewriteCond %{REQUEST_URI} !^/wp-content
RewriteCond %{REQUEST_URI} !^/wp-includes
RewriteCond %{REQUEST_URI} !^/\.well-known

RewriteRule ^(.*)$ https://www.investlife.es/?ref=comua [R=302,L,QSA]

# BEGIN LSCACHE

RewriteRule wp-content/.*/[^/]*(responsive|css|js|dynamic|loader|fonts)\.php - [E=cache-control:max-age=3600]

RewriteRule favicon\.ico$ - [E=cache-control:max-age=86400]
